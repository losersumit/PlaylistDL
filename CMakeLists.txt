cmake_minimum_required(VERSION 3.20)
project(PlaylistDL LANGUAGES CXX)
add_executable(PlaylistDL src/main.cpp)
target_compile_features(PlaylistDL PRIVATE cxx_std_17)
target_compile_definitions(PlaylistDL PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(PlaylistDL PRIVATE Shell32 Ole32)
if(MSVC)
  target_compile_options(PlaylistDL PRIVATE /W4 /O2 /utf-8)
else()
  target_compile_options(PlaylistDL PRIVATE -Wall -Wextra -O2)
endif()
